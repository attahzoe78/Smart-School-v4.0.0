   103→      <Card>

   109→              <Input value={search} onChange={(e) => setSearch(e.target.value)} placeholder="Search by name or admission no..." className="pl-9" />

   120→              <SelectContent>
   121→                <SelectItem value="">All Genders</SelectItem>
   122→                <SelectItem value="Male">Male</SelectItem>
   123→                <SelectItem value="Female">Female</SelectItem>

   130→            <div className="py-12 text-center text-sm text-muted-foreground">Loading students...</div>

   132→            <EmptyState

   150→                      <TableHead className="text-right">Actions</TableHead>
   151→                    </TableRow>
   152→                  </TableHeader>
   153→                  <TableBody>
   154→                    {paginatedStudents.map((s: any) => (
   155→                      <TableRow key={s.id} className="hover:bg-muted/50">
   156→                        <TableCell>
   157→                          <div className="flex items-center gap-3">
   158→                            <Avatar className="h-9 w-9">
   159→                              <AvatarImage src={s.photo || undefined} />

   165→                            </div>
   166→                          </div>
   167→                        </TableCell>
   168→                        <TableCell className="font-mono text-xs">{s.admissionNo}</TableCell>
   169→                        <TableCell>{s.currentClass?.name || "—"} {s.section?.name ? `(${s.section.name})` : ""}</TableCell>

   171→                          <Badge variant={s.gender === "Male" ? "default" : "secondary"} className={s.gender === "Male" ? "bg-blue-500" : "bg-pink-500 text-white"}>

   231→      {/* View Sheet */}

   237→function StudentFormDialog({ open, onOpenChange, student, classes, houses }: any) {
   238→  const queryClient = useQueryClient();
   239→  const [saving, setSaving] = useState(false);
   240→  const isEdit = !!student;
   241→
   242→  const [form, setForm] = useState<any>(() => student || {
   243→    firstName: "", lastName: "", middleName: "", gender: "Male", dateOfBirth: "",
   244→    bloodGroup: "", nationality: "Nigerian", stateOfOrigin: "Plateau",
   245→    admissionDate: new Date().toISOString().split("T")[0],
   246→    phone: "", email: "", address: "", city: "Jos", state: "Plateau",
   247→    currentClassId: "", sectionId: "", houseId: "",
   248→    religion: "Christianity", rollNo: "",
   249→    parent: { firstName: "", lastName: "", relation: "Father", phone: "", email: "", occupation: "", address: "" },
   250→  });
   251→
   252→  // Reset form when dialog opens
   253→  useMemo(() => {
   254→    if (open) {
   255→      setForm(student || {
   256→        firstName: "", lastName: "", middleName: "", gender: "Male", dateOfBirth: "",
   257→        bloodGroup: "", nationality: "Nigerian", stateOfOrigin: "Plateau",
   258→        admissionDate: new Date().toISOString().split("T")[0],
   259→        phone: "", email: "", address: "", city: "Jos", state: "Plateau",
   260→        currentClassId: "", sectionId: "", houseId: "",
   261→        religion: "Christianity", rollNo: "",
   262→        parent: { firstName: "", lastName: "", relation: "Father", phone: "", email: "", occupation: "", address: "" },
   263→      });
   264→    }
   265→  }, [open, student]);
   266→
   267→  const update = (key: string, value: any) => setForm((f: any) => ({ ...f, [key]: value }));
   268→  const updateParent = (key: string, value: any) => setForm((f: any) => ({ ...f, parent: { ...f.parent, [key]: value } }));
   269→
   270→  const selectedClass = classes.find((c: any) => c.id === form.currentClassId);
   271→  const sections = selectedClass?.sections || [];
   272→
   273→  async function handleSave() {
   274→    if (!form.firstName || !form.lastName) { toast.error("First and last name are required"); return; }
   275→    setSaving(true);
   276→    try {
   277→      const payload = { ...form, dateOfBirth: form.dateOfBirth || null, ageAsOn: null };
   278→      if (isEdit) {
   279→        await api(`/api/students`, { method: "PUT", body: JSON.stringify({ id: student.id, ...payload }) });
   280→        toast.success("Student updated successfully");
   281→      } else {
   282→        await api(`/api/students`, { method: "POST", body: JSON.stringify(payload) });
   283→        toast.success("Student admitted successfully! Login: " + (form.admissionNo || "auto-generated").toLowerCase().replace(/\//g, ""));
   284→      }
   285→      queryClient.invalidateQueries({ queryKey: ["students"] });
   286→      queryClient.invalidateQueries({ queryKey: ["dashboard"] });
   287→      onOpenChange(false);
   288→    } catch (e: any) {
   289→      toast.error(e.message);
   290→    } finally {
   291→      setSaving(false);
   292→    }
   293→  }
   294→
   295→  return (
   296→    <Dialog open={open} onOpenChange={onOpenChange}>
   297→      <DialogContent className="max-w-3xl max-h-[90vh] overflow-y-auto">
   298→        <DialogHeader>
   299→          <DialogTitle>{isEdit ? "Edit Student" : "New Student Admission"}</DialogTitle>
   300→        </DialogHeader>
   301→        <div className="space-y-4">
   302→          {/* Personal Info */}
   303→          <div>
   304→            <h4 className="text-sm font-semibold mb-2 text-emerald-700">Personal Information</h4>
   305→            <div className="grid grid-cols-2 md:grid-cols-3 gap-3">
   306→              <div className="space-y-1"><Label>First Name <span className="text-destructive">*</span></Label><Input value={form.firstName} onChange={(e) => update("firstName", e.target.value)} /></div>
   307→              <div className="space-y-1"><Label>Middle Name</Label><Input value={form.middleName} onChange={(e) => update("middleName", e.target.value)} /></div>
   308→              <div className="space-y-1"><Label>Last Name <span className="text-destructive">*</span></Label><Input value={form.lastName} onChange={(e) => update("lastName", e.target.value)} /></div>
   309→              <div className="space-y-1">
   310→                <Label>Gender</Label>
   311→                <Select value={form.gender} onValueChange={(v) => update("gender", v)}>
   312→                  <SelectTrigger><SelectValue /></SelectTrigger>
   313→                  <SelectContent><SelectItem value="Male">Male</SelectItem><SelectItem value="Female">Female</SelectItem></SelectContent>
   314→                </Select>
   315→              </div>
   316→              <div className="space-y-1"><Label>Date of Birth</Label><Input type="date" value={form.dateOfBirth} onChange={(e) => update("dateOfBirth", e.target.value)} /></div>
   317→              <div className="space-y-1">
   318→                <Label>Blood Group</Label>
   319→                <Select value={form.bloodGroup} onValueChange={(v) => update("bloodGroup", v)}>
   320→                  <SelectTrigger><SelectValue placeholder="Select" /></SelectTrigger>
   321→                  <SelectContent>{BLOOD_GROUPS.map((b) => <SelectItem key={b} value={b}>{b}</SelectItem>)}</SelectContent>

   422→  const [detail, setDetail] = useState<any>(null);
   423→
   424→  const { data, isLoading } = useQuery({
   425→    queryKey: ["student-detail", student?.id],
   426→    queryFn: () => api(`/api/students?id=${student.id}`),
   427→    enabled: !!student,
   428→  });
   429→
   430→  if (!student) return null;
   431→  const s = data || student;
   432→
   433→  return (
   434→    <Sheet open={!!student} onOpenChange={(o) => !o && onClose()}>
   435→      <SheetContent className="w-full sm:max-w-2xl overflow-y-auto p-0">
   436→        <SheetHeader className="p-4 border-b bg-muted/30">
   437→          <div className="flex items-center gap-4">
   438→            <Avatar className="h-16 w-16">
   439→              <AvatarImage src={s.photo || undefined} />
   440→              <AvatarFallback className="bg-emerald-600 text-white text-xl">{getInitials(s.firstName, s.lastName)}</AvatarFallback>
   441→            </Avatar>
   442→            <div className="flex-1">
   443→              <SheetTitle className="text-xl">{getFullName(s)}</SheetTitle>
   444→              <p className="text-sm text-muted-foreground">{s.admissionNo} • {s.currentClass?.name} {s.section?.name}</p>
   445→              <div className="flex items-center gap-2 mt-1">
   446→                <Badge variant={s.isActive ? "default" : "destructive"} className={s.isActive ? "bg-emerald-500" : ""}>{s.isActive ? "Active" : "Inactive"}</Badge>
   447→                <Badge variant="secondary">{s.gender}</Badge>
   448→              </div>
   449→            </div>
   450→          </div>
   451→        </SheetHeader>
   452→
   453→        <Tabs value={activeTab} onValueChange={setActiveTab} className="w-full">
   454→          <TabsList className="w-full justify-start px-4 border-b rounded-none bg-transparent h-auto p-0">
   455→            <TabsTrigger value="overview" className="rounded-none border-b-2 border-transparent data-[state=active]:border-emerald-600">Overview</TabsTrigger>
   456→            <TabsTrigger value="attendance" className="rounded-none border-b-2 border-transparent data-[state=active]:border-emerald-600">Attendance</TabsTrigger>
   457→            <TabsTrigger value="fees" className="rounded-none border-b-2 border-transparent data-[state=active]:border-emerald-600">Fees</TabsTrigger>
   458→            <TabsTrigger value="timeline" className="rounded-none border-b-2 border-transparent data-[state=active]:border-emerald-600">Timeline</TabsTrigger>
   459→          </TabsList>
   460→
   461→          <div className="p-4">
   462→            {isLoading ? (
   463→              <div className="py-8 text-center text-sm text-muted-foreground">Loading...</div>
   464→            ) : (
   465→              <>
   466→                <TabsContent value="overview" className="mt-0 space-y-4">
   467→                  <div className="grid grid-cols-2 gap-3">
   468→                    <InfoItem icon={Calendar} label="Date of Birth" value={`${formatDate(s.dateOfBirth)} (${calculateAge(s.dateOfBirth)} yrs)`} />
   469→                    <InfoItem icon={Activity} label="Blood Group" value={s.bloodGroup || "—"} />
   470→                    <InfoItem icon={MapPin} label="State of Origin" value={`${s.stateOfOrigin || "—"}, ${s.lga || ""}`} />
   471→                    <InfoItem icon={BookOpen} label="Religion" value={s.religion || "—"} />
   472→                    <InfoItem icon={Phone} label="Phone" value={s.phone || "—"} />
   473→                    <InfoItem icon={Mail} label="Email" value={s.email || "—"} />
   474→                    <InfoItem icon={MapPin} label="Address" value={`${s.address || "—"}, ${s.city || ""}`} />
   475→                    <InfoItem icon={Calendar} label="Admission Date" value={formatDate(s.admissionDate)} />
   476→                  </div>
   477→                  {s.house && <InfoItem icon={Users} label="House" value={s.house.name} />}
   478→                  {s.parent && (
   479→                    <div className="mt-4 pt-4 border-t">
   480→                      <h4 className="text-sm font-semibold mb-2">Parent / Guardian</h4>
   481→                      <div className="grid grid-cols-2 gap-3">
   482→                        <InfoItem icon={Users} label="Name" value={getFullName(s.parent)} />
   483→                        <InfoItem icon={Users} label="Relation" value={s.parent.relation} />
   484→                        <InfoItem icon={Phone} label="Phone" value={s.parent.phone || "—"} />
   485→                        <InfoItem icon={Mail} label="Email" value={s.parent.email || "—"} />
   486→                      </div>
   487→                    </div>
   488→                  )}
   489→                </TabsContent>
   490→
   491→                <TabsContent value="attendance" className="mt-0">
   492→                  <div className="space-y-2 max-h-96 overflow-y-auto">
   493→                    {s.attendance?.length ? s.attendance.map((a: any) => (
   494→                      <div key={a.id} className="flex items-center justify-between p-2 rounded-lg border">
   495→                        <span className="text-sm">{formatDate(a.date)}</span>
   496→                        <Badge variant={a.status === "Present" ? "default" : "destructive"} className={a.status === "Present" ? "bg-emerald-500" : a.status === "Late" ? "bg-amber-500" : ""}>{a.status}</Badge>
   497→                      </div>
   498→                    )) : <EmptyState title="No attendance records" />}
   499→                  </div>
   500→                </TabsContent>
   501→
   502→                <TabsContent value="fees" className="mt-0">
   503→                  <div className="space-y-2">
   504→                    {s.invoices?.length ? s.invoices.map((inv: any) => (
   505→                      <div key={inv.id} className="p-3 rounded-lg border">
   506→                        <div className="flex items-center justify-between">
   507→                          <span className="text-sm font-medium">{inv.invoiceNo}</span>
   508→                          <Badge variant={inv.status === "Paid" ? "default" : inv.status === "Partial" ? "secondary" : "destructive"} className={inv.status === "Paid" ? "bg-emerald-500" : inv.status === "Partial" ? "bg-amber-500 text-white" : ""}>{inv.status}</Badge>
   509→                        </div>
   510→                        <div className="grid grid-cols-3 gap-2 mt-2 text-xs">
   511→                          <div><span className="text-muted-foreground">Amount:</span> {formatCurrency(inv.amount)}</div>
   512→                          <div><span className="text-muted-foreground">Paid:</span> {formatCurrency(inv.paidAmount)}</div>
   513→                          <div><span className="text-muted-foreground">Balance:</span> {formatCurrency(inv.balance)}</div>
   514→                        </div>
   515→                      </div>
   516→                    )) : <EmptyState title="No fee records" />}
   517→                  </div>
   518→                </TabsContent>
   519→
   520→                <TabsContent value="timeline" className="mt-0">
   521→                  <div className="space-y-3">
   522→                    {s.timeline?.length ? s.timeline.map((t: any) => (
   523→                      <div key={t.id} className="flex gap-3 p-3 rounded-lg border">
   524→                        <div className="flex h-8 w-8 items-center justify-center rounded-full bg-emerald-100 text-emerald-600 shrink-0">
   525→                          <FileText className="h-4 w-4" />
   526→                        </div>
   527→                        <div>
   528→                          <p className="text-sm font-medium">{t.title}</p>
   529→                          <p className="text-xs text-muted-foreground">{formatDate(t.date)}</p>
   530→                          {t.description && <p className="text-sm mt-1">{t.description}</p>}
   531→                        </div>
   532→                      </div>
   533→                    )) : <EmptyState title="No timeline events" />}
   534→                  </div>
   535→                </TabsContent>
   536→              </>
   537→            )}
   538→          </div>
   539→        </Tabs>
   540→      </SheetContent>
   541→    </Sheet>
   542→  );
   543→}
   544→
   545→function InfoItem({ icon: Icon, label, value }: any) {

   547→    <div className="flex items-start gap-2">
   548→      <Icon className="h-4 w-4 text-muted-foreground mt-0.5 shrink-0" />
   549→      <div>
   550→        <p className="text-xs text-muted-foreground">{label}</p>
   551→        <p className="text-sm font-medium">{value}</p>
   552→      </div>
   553→    </div>
   554→  );
   555→}
